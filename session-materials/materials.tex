% One A6 action card. #1 column (0-1), #2 row (0-1), #3 icon, #4 Portuguese, #5 English.
\newcommand{\actioncard}[5]{%
  \begin{scope}[shift={($(current page.north west)+(#1*148.5mm,-#2*105mm)$)}]
    \node[text=coral] at (74.25mm,-30mm) {\fontsize{60}{60}\selectfont #3};
